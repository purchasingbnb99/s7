// ========================================== // FUNGSI UTAMA DO GET & DO POST // ========================================== function doGet(e) { const action = e.parameter.action; if (action === 'getBerkas') return createJsonResponse(getBerkasData()); if (action === 'getSample') return createJsonResponse(getSampleData()); if (action === 'getLogs') return createJsonResponse(getLogsData()); if (action === 'getPoKembali') { return createJsonResponse(getPoKembaliData(e.parameter.month, e.parameter.year)); } return createJsonResponse({ status: "Ready" }); } function doPost(e) { const data = JSON.parse(e.postData.contents); const action = data.action; try { // AUTHENTICATION if (action === 'verifyAdminPassword') { return createJsonResponse({ success: data.password === 'admin123' }); } // MODULE BERKAS if (action === 'createBerkasBatch') return createJsonResponse(createBerkasBatch(data.items)); if (action === 'updateDocumentDetails') return createJsonResponse(updateDocumentDetails(data)); if (action === 'updatePenerimaStatus') return createJsonResponse(updatePenerimaStatus(data)); if (action === 'unlockDocument') return createJsonResponse(unlockDocument(data.id)); if (action === 'deleteDocument') return createJsonResponse(deleteDocument(data.id)); if (action === 'deleteGroupDate') return createJsonResponse(deleteGroupDate(data.ids)); // MODULE SAMPLE if (action === 'createSample' || action === 'updateSample') return createJsonResponse(saveSample(data)); if (action === 'deleteSample') return createJsonResponse(deleteSample(data.id)); // MODULE PO KEMBALI if (action === 'createPoKembali' || action === 'updatePoKembali') return createJsonResponse(savePoKembali(data)); if (action === 'deletePoKembali') return createJsonResponse(deletePoKembali(data.id)); if (action === 'executeYearlyBackup') return createJsonResponse(processYearlyBackup()); // LOGBOOK if (action === 'addLogbookEntry') return createJsonResponse(addLogbookEntry(data.logData)); if (action === 'clearLogbook') return createJsonResponse(clearLogbook()); return createJsonResponse({ success: false, message: "Action not found" }); } catch (error) { return createJsonResponse({ success: false, error: error.message }); } } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); } // ========================================== // HELPER: MANAJEMEN SHEET // ========================================== function getSheet(sheetName, headers) { let sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) { sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet(sheetName); sheet.appendRow(headers); } return sheet; } // ========================================== // LOGIKA MODULE BERKAS // ========================================== function getBerkasData() { const sheet = getSheet('Berkas', ['ID', 'TANGGAL', 'NO_PO', 'VENDOR', 'NOMINAL', 'NOTES', 'IS_CHECKED', 'IS_LOCKED']); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; return data.slice(1).map(row => ({ id: row[0], tanggal: row[1], noPo: row[2], vendor: row[3], nominal: row[4], notes: row[5], isChecked: row[6] === true || row[6] === 'true', isLocked: row[7] === true || row[7] === 'true' })); } function createBerkasBatch(items) { const sheet = getSheet('Berkas', []); items.forEach(item => { sheet.appendRow([item.id, item.tanggal, item.noPo, item.namaVendor, item.nominal, '', false, false]); }); return { success: true }; } function updateDocumentDetails(data) { const sheet = getSheet('Berkas', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === data.id) { sheet.getRange(i + 1, 2, 1, 4).setValues([[data.tanggal, data.noPo, data.namaVendor, data.nominal]]); return { success: true }; } } return { success: false }; } function updatePenerimaStatus(data) { const sheet = getSheet('Berkas', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === data.id) { sheet.getRange(i + 1, 6, 1, 3).setValues([[data.notes, data.isCompleted, data.isCompleted]]); return { success: true }; } } return { success: false }; } function unlockDocument(id) { const sheet = getSheet('Berkas', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === id) { sheet.getRange(i + 1, 8).setValue(false); return { success: true }; } } return { success: false }; } function deleteDocument(id) { const sheet = getSheet('Berkas', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === id) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } function deleteGroupDate(ids) { const sheet = getSheet('Berkas', []); const rows = sheet.getDataRange().getValues(); for (let i = rows.length - 1; i > 0; i--) { if (ids.includes(rows[i][0])) sheet.deleteRow(i + 1); } return { success: true }; } // ========================================== // LOGIKA MODULE SAMPLE // ========================================== function getSampleData() { const sheet = getSheet('Sample', ['ID', 'TANGGAL', 'BARANG', 'TUJUAN', 'KETERANGAN', 'URL_FOTO']); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; return data.slice(1).map(row => ({ id: row[0], tanggal: row[1], namaBarang: row[2], tujuan: row[3], keterangan: row[4], urlFoto: row[5] })); } function saveSample(payload) { const sheet = getSheet('Sample', []); if (payload.action === 'updateSample') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 2, 1, 5).setValues([[payload.tanggal, payload.namaBarang, payload.tujuan, payload.keterangan, payload.urlFoto]]); return { success: true }; } } } sheet.appendRow([payload.id, payload.tanggal, payload.namaBarang, payload.tujuan, payload.keterangan, payload.urlFoto]); return { success: true }; } function deleteSample(id) { const sheet = getSheet('Sample', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === id) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } // ========================================== // LOGIKA MODULE PO KEMBALI // ========================================== function getPoKembaliData(month, year) { const sheet = getSheet('PO_Kembali', ['ID', 'TAHUN', 'BULAN', 'TGL_KIRIM', 'TGL_KEMBALI', 'PREFIX', 'NO_PO', 'TGL_PO', 'NAMA_VENDOR', 'CATATAN']); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; let results = []; for (let i = 1; i < data.length; i++) { const row = data[i]; const rowYear = String(row[1]); const rowMonth = String(row[2]); // Server-Side Filter if ((!year || rowYear === String(year)) && (!month || rowMonth === String(month))) { results.push({ id: row[0], tahun: row[1], bulan: row[2], tglKirim: row[3], tglKembali: row[4], kodePrefix: row[5], noPo: row[6], tglPo: row[7], namaVendor: row[8], catatan: row[9] }); } } return results; } function savePoKembali(payload) { const sheet = getSheet('PO_Kembali', []); let dateParts = payload.tglKirim.split('-'); let tahun = dateParts.length === 3 ? dateParts[0] : new Date().getFullYear(); let bulan = dateParts.length === 3 ? dateParts[1] : ('0' + (new Date().getMonth() + 1)).slice(-2); if (payload.action === 'updatePoKembali') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 1, 1, 10).setValues([[ payload.id, tahun, bulan, payload.tglKirim, payload.tglKembali, payload.kodePrefix, payload.noPo, payload.tglPo, payload.namaVendor, payload.catatan || '' ]]); return { success: true }; } } } sheet.appendRow([ payload.id, tahun, bulan, payload.tglKirim, payload.tglKembali, payload.kodePrefix, payload.noPo, payload.tglPo, payload.namaVendor, payload.catatan || '' ]); return { success: true }; } function deletePoKembali(id) { const sheet = getSheet('PO_Kembali', []); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === id) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } // ========================================== // LOGIKA BACKUP & PURGE // ========================================== function processYearlyBackup() { const sheet = getSheet('PO_Kembali', []); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return { success: true, message: "Data kosong, tidak perlu backup." }; let csvContent = ""; data.forEach(row => { let cleanRow = row.map(cell => `"${String(cell).replace(/"/g, '""')}"`); csvContent += cleanRow.join(",") + "\n"; }); const backupFileName = "Backup_PO_Kembali_" + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd") + ".csv"; let folders = DriveApp.getFoldersByName("Purchasing"); let targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder("Purchasing"); let file = targetFolder.createFile(backupFileName, csvContent, MimeType.CSV); const totalRows = sheet.getMaxRows(); if (totalRows > 2) { sheet.deleteRows(2, totalRows - 2); sheet.getRange(2, 1, 1, sheet.getLastColumn()).clearContent(); } return { success: true, downloadUrl: file.getDownloadUrl(), message: "Backup berhasil disimpan di cloud dan data lama telah dibersihkan." }; } // ========================================== // LOGIKA LOGBOOK // ========================================== function getLogsData() { const sheet = getSheet('Logs', ['ID', 'TIMESTAMP', 'ROLE', 'ACTION', 'DETAILS']); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; return data.slice(1).map(row => ({ id: row[0], timestamp: row[1], role: row[2], action: row[3], details: row[4] })).reverse(); } function addLogbookEntry(log) { const sheet = getSheet('Logs', []); sheet.appendRow([log.id, log.timestamp, log.role, log.action, log.details]); return { success: true }; } function clearLogbook() { const sheet = getSheet('Logs', []); const totalRows = sheet.getMaxRows(); if (totalRows > 2) { sheet.deleteRows(2, totalRows - 2); sheet.getRange(2, 1, 1, sheet.getLastColumn()).clearContent(); } return { success: true }; }
